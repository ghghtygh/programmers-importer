package programmers.p42;

import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.Arguments;
import org.junit.jupiter.params.provider.MethodSource;
import java.util.stream.Stream;

import static org.junit.jupiter.api.Assertions.assertArrayEquals;

class SolutionTest {

    @ParameterizedTest(name = "공개 예제 {index}")
    @MethodSource("examples")
    void matchesExample(int[][] grid, int[][] expected) throws Exception {
        assertArrayEquals(expected, new Solution().solution(grid));
    }

    static Stream<Arguments> examples() {
        return Stream.of(
            Arguments.of(new Object[]{new int[][]{new int[]{1, 2}, new int[]{3, 4}}, new int[][]{new int[]{1, 2}, new int[]{3, 4}}})
        );
    }
}
